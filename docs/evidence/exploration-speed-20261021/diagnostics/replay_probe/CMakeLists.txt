cmake_minimum_required(VERSION 3.8)
project(slam_replay_probe)
find_package(ament_cmake REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(pluginlib REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization)
add_executable(replay replay.cpp)
ament_target_dependencies(replay slam_toolbox pluginlib)
get_filename_component(ROS_PREFIX "${slam_toolbox_DIR}/../../.." ABSOLUTE)
find_library(KARTO_LIBRARY kartoSlamToolbox HINTS "${ROS_PREFIX}/lib" REQUIRED)
find_library(TOOLBOX_LIBRARY toolbox_common HINTS "${ROS_PREFIX}/lib" REQUIRED)
target_link_libraries(replay ${TOOLBOX_LIBRARY} ${KARTO_LIBRARY} Boost::serialization tbb)
