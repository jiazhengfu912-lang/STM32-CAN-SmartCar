cmake_minimum_required(VERSION 3.8)
project(smartcar_nav2)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav2_controller REQUIRED)
find_package(nav2_planner REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav_2d_utils REQUIRED)
find_package(tf2_ros REQUIRED)
foreach(kind controller planner)
  add_executable(safe_${kind}_server src/safe_${kind}_server.cpp)
  target_compile_features(safe_${kind}_server PRIVATE cxx_std_17)
  ament_target_dependencies(safe_${kind}_server rclcpp nav2_controller nav2_planner
                            nav2_costmap_2d nav_2d_utils tf2_ros)
  install(TARGETS safe_${kind}_server DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
install(DIRECTORY ../../maps/office_01 DESTINATION share/${PROJECT_NAME}/maps)
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/navigation_gate.py scripts/navigate.py scripts/check_navigation.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/gate_core.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(gate_core tests/test_gate_core.py)
  ament_add_pytest_test(planning_clearance tests/test_planning_clearance.py)
  ament_add_pytest_test(recovery_policy tests/test_recovery_policy.py)
  ament_add_pytest_test(controller_runtime tests/test_controller_runtime.py TIMEOUT 90
    ENV SMARTCAR_TEST_CONTROLLER=$<TARGET_FILE:safe_controller_server>)
endif()
ament_package()
